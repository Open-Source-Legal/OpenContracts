    Open-Vocabulary Authority Discovery — Phase 0 + 1 Implementation Plan¶

    For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

    Goal: Make the reference-enrichment engine open-vocabulary — detect citations to authorities outside the static registry (US Code, CFR, Federal Register, state codes, municipal codes, bare Acts), classified by jurisdiction and authority type, and surface them through a new discover_authorities agent tool — without regressing the trusted Tier-1 registry path.

    Architecture: Phase 0 adds two classification dimensions (jurisdiction, authority_type) to CorpusReference and a queryable AuthorityNamespace registry table seeded from today's static AUTHORITY_PREFIX map, then makes authority_alias_registry() read the table. Phase 1 adds a deterministic, shape-based GenericCitationExtractor (Tier-2a) running alongside the registry extractor, a reconciliation step that lets the registry win on overlap, and persistence of the new classification + detection provenance on every reference. Everything is additive: with the default tier set the engine behaves exactly as today.

    Tech Stack: Django 4.x, PostgreSQL, Python 3, pytest / SimpleTestCase / TestCase, the existing opencontractserver/enrichment/ engine.

    Design spec: docs/superpowers/specs/2026-06-14-open-vocabulary-authority-discovery-design.md (Components 1, 2; Phases 0, 1 in §15).

    class AuthorityNamespace(django.db.models.Model):
    """Registry of legal-authority bodies (one row per canonical_key prefix).

    Reference data, not user-owned content — so a plain ``Model``: a global
    registry wants neither a required ``creator`` nor guardian permission rows.
    Seeded from ``constants.AUTHORITY_PREFIX`` (Phase 0) and extended as new
    bodies of law are discovered. ``authority_alias_registry`` reads this table
    to drive Tier-1 extraction without a code change.
    """

    prefix = django.db.models.CharField(max_length=64, unique=True, db_index=True)
    display_name = django.db.models.CharField(max_length=255)
    jurisdiction = django.db.models.CharField(
        max_length=64, null=True, blank=True, db_index=True
    )
    authority_type = django.db.models.CharField(
        max_length=32, null=True, blank=True, db_index=True
    )
    # Surface forms seen in text (lowercased), fed into the extractor alias map.
    aliases = django.db.models.JSONField(default=list, blank=True)
    # Routing / provenance (populated by later phases; nullable now).
    provider = django.db.models.CharField(max_length=64, null=True, blank=True)
    source_root_url = django.db.models.URLField(max_length=500, null=True, blank=True)
    license = django.db.models.CharField(max_length=64, null=True, blank=True)
    # Global namespaces always contribute aliases; corpus-linked ones only when
    # the corpus is visible (wired in authority_alias_registry).
    is_global = django.db.models.BooleanField(default=True, db_index=True)
    authority_corpus = django.db.models.ForeignKey(
        "corpuses.Corpus",
        on_delete=django.db.models.SET_NULL,
        null=True,
        blank=True,
        related_name="authority_namespaces",
    )
    created = django.db.models.DateTimeField(auto_now_add=True)
    modified = django.db.models.DateTimeField(auto_now=True)

    class Meta:
        indexes = [
            django.db.models.Index(fields=["jurisdiction", "authority_type"]),
        ]

    def __str__(self) -> str:
        return f"AuthorityNamespace({self.prefix}: {self.display_name})"

# bare-Act detection (Task 11) is intentionally lower.
_CONF_STRUCTURED = 0.9

# A section token: digits, optional trailing letter, optional (a)(2) subsections,
# optional hyphenated rule tail (10b-5, 261.4).
_SEC = r"\d+[A-Za-z]?(?:\.\d+)?(?:[-–]\d+)?(?:\([0-9a-zA-Z]+\))*"

_STAT_RE = re.compile(r"\b(?P<vol>\d+)\s+Stat\.?\s+(?P<page>\d[\d,]*)")

    return Candidate(
        reference_type=C.REF_LAW,
        start=start,
        end=end,
        raw_text=raw,
        canonical_key=key,
        normalized_data={"authority": key.split(":", 1)[0], "tier": "grammar"},
        jurisdiction=jur,
        authority_type=typ,
        detection_tier=C.DETECTION_TIER_GRAMMAR,
        detection_confidence=conf,

    """Merge detection layers with precedence: registry > grammar > llm.

A higher-precedence candidate suppresses any lower or equal-precedence
candidate whose span overlaps it. Within a single layer, the earlier span (by
start offset) wins an overlap. The Tier-1 (registry) list is always preserved
in full — callers pass it as ``primary``.
"""

from __future__ import annotations

from opencontractserver.enrichment.extractor import Candidate


def _overlaps(a: tuple[int, int], b: tuple[int, int]) -> bool:
    return not (a[1] <= b[0] or a[0] >= b[1])


def reconcile(
    primary: list[Candidate], secondary: list[Candidate]
) -> list[Candidate]:
    """Keep all ``primary``; add ``secondary`` candidates that don't overlap a
    kept span. ``secondary`` is processed in document order so the earliest
    span wins an intra-layer overlap."""
    kept: list[Candidate] = list(primary)
    occupied: list[tuple[int, int]] = [(c.start, c.end) for c in primary]
    for cand in sorted(secondary, key=lambda c: (c.start, c.end)):
        span = (cand.start, cand.end)
        if any(_overlaps(span, used) for used in occupied):
            continue
        kept.append(cand)
        occupied.append(span)
    return kept

            prefix = key.split(":", 1)[0]
            prefixes.add(prefix)
            entry = by_key.setdefault(
                key,
                {
                    "canonical_key": key,
                    "prefix": prefix,
                    "jurisdiction": cand.jurisdiction,
                    "authority_type": cand.authority_type,
                    "detection_tier": cand.detection_tier,
                    "mention_count": 0,
                },
            )
            entry["mention_count"] += 1
            if cand.jurisdiction:
                by_jurisdiction[cand.jurisdiction] += 1
            if cand.authority_type:
                by_type[cand.authority_type] += 1

        known = set(
            AuthorityNamespace.objects.filter(prefix__in=prefixes).values_list(
                "prefix", flat=True
            )
        )
        new_namespaces = [
            {"prefix": p, "jurisdiction": None, "authority_type": None}
            for p in sorted(prefixes - known)
        ]
        # Enrich new-namespace classification from any detected key for that prefix.
        prefix_class = {
            e["prefix"]: (e["jurisdiction"], e["authority_type"])
            for e in by_key.values()
        }
        for n in new_namespaces:
            n["jurisdiction"], n["authority_type"] = prefix_class.get(
                n["prefix"], (None, None)
            )

        return {
            "corpus_id": corpus_id,
            "documents_scanned": len(documents),
            "total_candidates": len(resolutions),
            "by_key": by_key,
            "by_jurisdiction": dict(by_jurisdiction),
            "by_authority_type": dict(by_type),
            "new_namespaces": new_namespaces,
        }
